    std::vector<std::vector<Token>> lines;
    std::vector<Token> tokens;
    
    // Explicit state machine
    enum class State { START, IDENT, NUMBER };
    State state = State::START;
    
    int start = 0;
    int line = 1;
    int col = 1;
    size_t i = 0;
    
    std::map<std::string, std::string> KEYWORDS = {
        {"i32", "keyword"}, {"mut", "keyword"}, {"exit", "keyword"}
    };

    while (i <= data.length()) { 
        // Handle the extra end-of-input step defined in the skeleton
        bool is_eof = (i == data.length());
        unsigned char b = is_eof ? 0 : data[i];

        if (state == State::START) {
            if (is_eof) {
                break;
            } else if (b == ' ' || b == '\t') {
                // pass: spaces and tabs are dropped
            } else if (std::isalpha(b)) {
                state = State::IDENT;
                start = i;
            } else if (b == '\n') {
                lines.push_back(tokens);
                tokens.clear();
                line += 1;
                col = 0; // Resets to 0, will become 1 after col += 1 at the loop's end
            } else if (std::isdigit(b)) {
                state = State::NUMBER;
                start = i;
            } else if (b == '{') {
                tokens.push_back({"block", "{", line, col});
            } else {
                throw std::runtime_error("line " + std::to_string(line) + ":" + std::to_string(col) + ": unexpected byte '" + std::string(1, b) + "'");
            }
        } 
        else if (state == State::IDENT) {
            // Identifiers are letters, digits and _, not starting with a digit
            if (!is_eof && (std::isalpha(b) || std::isdigit(b) || b == '_')) { 
                // pass
            } else {
                std::string word = data.substr(start, i - start);
                std::string kind = KEYWORDS.count(word) ? KEYWORDS[word] : "identifier";
                // Calculate the original starting column of the word
                int token_col = col - (i - start); 
                tokens.push_back({kind, word, line, token_col});
                state = State::START;
                continue; // continue without advancing to re-read this byte in START
            }
        } 
        else if (state == State::NUMBER) {
            if (!is_eof && std::isdigit(b)) {
                // pass
            } else {
                std::string word = data.substr(start, i - start);
                int token_col = col - (i - start);
                tokens.push_back({"number", word, line, token_col});
                state = State::START;
                continue; 
            }
        }

        col += 1;
    }
    
    // Append the last line if there is no trailing newline

        lines.push_back(tokens);
    }

**Required Fixes**

* **Missing State Reset:** Inside `State::ASSIGNMENT`, you forgot to reset `state = State::START;` after emitting `:=`. The lexer would get stuck in the assignment state forever.

* **Assignment Error Message:** The assignment explicitly lists `a: not followed by =` as an error to report. The `ASSIGNMENT` fallback must throw this specific message instead of `unexpected byte`.


* **Non-ASCII Enforcement:** The rules dictate that any byte above 127 is an unknown byte. You must check `b > 127` in the `START` state.

auto resolveValue = [&](const Token& token) -> llvm::Value* {
    if (token.kind == "number") {
        // Generate an LLVM constant for the numeric value
        return llvm::ConstantInt::get(i32Type, std::stoi(token.text), true);
    } 
    else if (token.kind == "identifier") {
        // Check if the variable exists in the symbol table
        if (symbols.find(token.text) == symbols.end()) {
